    <!-- JAVASCRIPT CONNECTIVITY TO NODE.JS BACKEND -->
    <script>
        document.getElementById('loanForm').addEventListener('submit', async function(e) {
            e.preventDefault();

            const formData = {
                fullName: document.getElementById('fullName').value,
                cnic: document.getElementById('cnic').value,
                mobile: document.getElementById('mobile').value,
                gender: document.getElementById('gender').value,
                dob: document.getElementById('dob').value,
                province: document.getElementById('province').value,
                address: document.getElementById('address').value
            };

            const submitBtn = this.querySelector('button[type="submit"]');
            const origText = submitBtn.innerHTML;
            submitBtn.disabled = true;
            submitBtn.innerHTML = `<i class="fa-solid fa-spinner animate-spin"></i> Submitting...`;

            try {
                // Fetch call to your Node.js backend running on port 5000
                const response = await fetch('http://localhost:5000/api/apply-loan', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify(formData)
                });

                const result = await response.json();

                if (result.success) {
                    showModal('success', result.message, `Reference ID: ${result.referenceId}`);
                    document.getElementById('loanForm').reset();
                } else {
                    showModal('error', result.message || 'Kuch masla hua hai. Dobara koshish karein.');
                }
            } catch (err) {
                // Server disconnect fallback alert
                showModal('error', 'Server Connection Failed! Computer par "node server.js" run karna zaroori hai.');
            } finally {
                submitBtn.disabled = false;
                submitBtn.innerHTML = origText;
            }
        });

        function showModal(type, msg, ref = '') {
            const modal = document.getElementById('statusModal');
            const icon = document.getElementById('modalIcon');
            const title = document.getElementById('modalTitle');
            const message = document.getElementById('modalMessage');
            const refBox = document.getElementById('refBox');

            if (type === 'success') {
                icon.className = "w-16 h-16 rounded-full bg-emerald-100 text-emerald-600 flex items-center justify-center text-2xl mx-auto";
                icon.innerHTML = `<i class="fa-solid fa-check"></i>`;
                title.innerText = "Darkhwast Mosool Ho Chuki Hai";
                message.innerText = msg;
                if (ref) {
                    refBox.innerText = ref;
                    refBox.classList.remove('hidden');
                }
            } else {
                icon.className = "w-16 h-16 rounded-full bg-rose-100 text-rose-600 flex items-center justify-center text-2xl mx-auto";
                icon.innerHTML = `<i class="fa-solid fa-triangle-exclamation"></i>`;
